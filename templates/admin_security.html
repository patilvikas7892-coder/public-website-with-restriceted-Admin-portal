Cloud Infrastructure as Code (Terraform / Nginx)

Terraform Security Group (security_groups.tf)

resource "aws_security_group" "web_and_admin_sg" {
  name        = "assignment-19-portal-sg"
  description = "Allows public web and restricts SSH and admin to College IP"
  vpc_id      = aws_vpc.main.id

  # Public HTTP (Port 80) -> Auto HTTPS Redirect
  ingress {
    from_port   = 80
    to_port     = 80
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  # Public HTTPS (Port 443)
  ingress {
    from_port   = 443
    to_port     = 443
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  # Restricted SSH (Port 22) -> College IP Only
  ingress {
    from_port   = 22
    to_port     = 22
    protocol    = "tcp"
    cidr_blocks = ["203.0.113.195/32"] # College Admin IP
  }
}

Nginx Admin IP Whitelist (nginx.conf)

server {
    listen 443 ssl http2;
    server_name cloudtech.edu;

    # Public Homepage and Static S3 Assets
    location / {
        proxy_pass http://127.0.0.1:5000;
        allow all;
    }

    # Strict Admin Portal IP Whitelist
    location /admin {
        # Allow College Campus Dedicated IP
        allow 203.0.113.195/32;
        # Allow College Campus Subnet
        allow 203.0.113.0/24;
        # Deny all external public internet IPs
        deny all;

        proxy_pass http://127.0.0.1:5000;
    }
}
